Code Examples

Initialize SDK 
import { RiviumSync } from '@rivium/sync-web';

await RiviumSync.init({
  apiKey: 'your-api-key',
  offlineEnabled: true,
  debugMode: true,
});

const db = RiviumSync.database('my-app');
const collection = db.collection('todos');
CRUD Operations 
// Create
const doc = await collection.add({
  title: 'My Task',
  completed: false,
});

// Read
const doc = await collection.get('doc-id');

// Update
await collection.update('doc-id', { completed: true });

// Delete
await collection.delete('doc-id');
Queries 
// Query with filters
const results = await collection.query({
  filters: [
    { field: 'completed', operator: '==', value: false },
    { field: 'priority', operator: '>=', value: 5 },
  ],
  orderBy: 'createdAt',
  orderDirection: 'desc',
  limit: 10,
});
Batch Operations 
const batch = db.batch();

batch.set(collection, 'doc1', { name: 'Alice' });
batch.update(collection, 'doc2', { status: 'active' });
batch.delete(collection, 'doc3');

await batch.commit(); // Atomic
Realtime Listeners 
// Listen to collection changes
const unsubscribe = collection.onSnapshot((documents) => {
  console.log('Collection updated:', documents.length);
});

// Stop listening
unsubscribe();
